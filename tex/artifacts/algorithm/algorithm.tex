\begin{algorithm}[t]
\caption{Trace-corrected training of the ML-PIFLNN}
\label{alg:mlpiflnn}
\begin{algorithmic}[1]
\Require Fractional order $\alpha \in (1/2, 1]$; M\"untz sequence $\Lambda = \{\lambda_k\}_{k=0}^n$; drift $b$, diffusion $\sigma$, initial condition $y_0$; collocation size $N_c$; regularizing weights $\lambda_b, \lambda_\sigma > 0$; Gauss--Newton tolerance $\varepsilon_{\mathrm{GN}}$ and iteration cap $K_{\mathrm{GN}}$
\Ensure Trained, trace-corrected parameters $\hat{\boldsymbol \theta} = [\hat{\mathbf c}^\top \; \hat{\mathbf c}_b^\top \; \hat{\mathbf c}_\sigma^\top]^\top$

\Statex \textbf{Basis and operational matrix assembly}
\State Compute $\{\eta_k^{(m)}\}_{k=0}^m$ for $m = 0, \dots, n$ and form $\mathbf M_n^\Lambda(t) = [M_0^\Lambda(t) \; \cdots \; M_n^\Lambda(t)]^\top$ \Comment{\S\ref{sec:muntz_legendre}}
\State Assemble the monomial change-of-basis matrix $\mathbf C$, its inverse $\mathbf C^{-1}$, and the diagonal matrix $\mathbf D_\alpha = \diag \left ( \Gamma(\lambda_k+1)/\Gamma(\lambda_k+\alpha+1) \right )$; form $\mathbf P_\alpha(t) = t^\alpha \mathbf C \mathbf D_\alpha \mathbf C^{-1}$
\State Sample the driving path $B$ on a fine reference grid; assemble the static tensor $\mathbf T$, the cross-covariance $\boldsymbol \Xi$, and $\boldsymbol \Omega$; set $\mathbf S_\alpha = \boldsymbol \Xi \boldsymbol \Omega^{-1}$

\Statex \textbf{Collocation}
\State Generate the Chebyshev nodes $T_c = \{t_i\}_{i=1}^{N_c}$
\State Assemble $\boldsymbol \Phi$, $\boldsymbol \Phi_P$, $\boldsymbol \Phi_S$ at $T_c$

\Statex \textbf{First pass: uncorrected solve}
\If {$b, \sigma$ affine in $y$}
    \State Assemble $\mathbf K, \mathbf d$ \eqref{eq:block_system}
    \State $\boldsymbol \theta^* \gets \mathbf K^+ \mathbf d$ \Comment{Moore--Penrose pseudoinverse}
\Else
    \State Initialize $\boldsymbol \theta^{(0)}$ from the affine solution linearized about $y_0$
    \For{$k = 0, 1, \dots, K_{\mathrm{GN}} - 1$}
        \State $\boldsymbol \theta^{(k+1)} \gets \boldsymbol \theta^{(k)} - \xi_k \left ( \nabla_{\boldsymbol \theta} \mathbf r(\boldsymbol \theta^{(k)}) \right )^+ \mathbf r(\boldsymbol \theta^{(k)})$
        \If{$\norm{\boldsymbol \theta^{(k+1)} - \boldsymbol \theta^{(k)}} < \varepsilon_{\mathrm{GN}}$}
            \State \textbf{break}
        \EndIf
    \EndFor
    \State $\boldsymbol \theta^* \gets \boldsymbol \theta^{(k+1)}$
\EndIf

\Statex \textbf{Malliavin trace correction, computed once from $\boldsymbol \theta^*$ (\S\ref{sec:malliavin})}
\State Compute $D_s \boldsymbol \Phi_S = \mathbf g(s) \mathbf M_n^\Lambda(s)^\top$ \Comment{Corollary~\ref{cor:DPhiS}}
\State Compute $D_s \boldsymbol \theta^*$ via \eqref{eq:Dtheta} (general) or \eqref{eq:Dtheta_affine} (affine)
\State Extract $D_s \mathbf c_\sigma^* = \mathbf E_\sigma D_s \boldsymbol \theta^*$
\State Evaluate the frozen trace $\boldsymbol \tau = [\tau_n[\mathbf c_\sigma^*](t_1) \; \cdots \; \tau_n[\mathbf c_\sigma^*](t_{N_c})]^\top$, $\tau_n[\mathbf c_\sigma^*](t_i) = \int_0^{t_i} K_\alpha(t_i, s) (D_s \mathbf c_\sigma^*)^\top \mathbf M_n^\Lambda(s) \dd{s}$

\Statex \textbf{Second pass: trace-corrected re-solve (once, not iterated to self-consistency)}
\If {$b, \sigma$ affine in $y$}
    \State Re-solve $\hat{\boldsymbol \theta} \gets \mathbf K^+ \hat{\mathbf d}$ against the trace-adjusted target $\hat{\mathbf d} = \mathbf d - [\boldsymbol \tau^\top \; \mathbf 0^\top \; \mathbf 0^\top]^\top$ \eqref{eq:d_corrected}
\Else
    \State Replace the OME residual block by $\hat{\mathbf r}_1(\boldsymbol \theta) = \mathbf r_1(\boldsymbol \theta) + \boldsymbol \tau$ \eqref{eq:gn_corrected_residual}
    \State Warm-start from $\boldsymbol \theta^*$ and repeat the Gauss--Newton loop above on $\hat{\mathbf r} = [\hat{\mathbf r}_1^\top \; \mathbf r_2^\top \; \mathbf r_3^\top]^\top$ to convergence, giving $\hat{\boldsymbol \theta}$
\EndIf

\State \Return $\hat{\boldsymbol \theta}$
\end{algorithmic}
\end{algorithm}

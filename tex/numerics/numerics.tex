\begin{example}[Ornstein--Uhlenbeck Process]
\label{ex:linear_fractional_sde}
We first consider the general linear Caputo fractional stochastic differential equation with mean-reverting drift and additive diffusion:
\begin{equation}
    \DD y(t) = \theta(\mu - y(t)) + \sigma \sdv, \quad t \in [0, 1], \quad y(0) = y_0.
\end{equation}
\begin{enumerate}
    \item \textbf{Zero diffusion:} Let $\mu = \sigma = 0$, $\theta = 1.0$, and $y_0 = 1.0$. The equation reduces to the deterministic Caputo fractional differential equation $\DD y(t) = -y(t)$, with exact analytical solution $y(t) = E_\alpha(-t^\alpha)$.

\begin{example}[Geometric Brownian Motion]
\label{ex:fractional_gbm}
We consider the fractional Geometric Brownian Motion (GBM) characterized by linear drift and state-dependent multiplicative diffusion:
\begin{equation}
    \DD y(t) = \mu y(t) + \sigma y(t) \sdv, \quad t \in [0, 1], \quad y(0) = y_0.
\end{equation}
We set the drift coefficient $\mu = 0.3$, volatility $\sigma = 0.15$, and initial state $y_0 = 1.0$.

\begin{example}[Cox--Ingersoll--Ross Process]
\label{ex:fractional_cir}

\begin{equation}
    \DD y(t) = \mu y(t) + \sigma \sqrt{y(t)} \sdv, \quad t \in [0, 1], \quad y(0) = y_0.
\end{equation}
